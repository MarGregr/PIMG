CREATE EXTENSION IF NOT EXISTS postgis;

-- statystyki pojazdy
select 
extract(year from data_ostatniej_rejestracji_w_kraju) as rok, 
count(p.*) as liczba,
case p.rodzaj_pojazdu when 'CI¥GNIK SAMOCHODOWY' THEN 'SAMOCHÓD CIÊ¯AROWY' when 'SAMOCHÓD SPECJALNY' then 'SAMOCHÓD CIÊ¯AROWY' when 'SAMOCHODOWY INNY' THEN 'SAMOCHÓD CIÊ¯AROWY' when 'SAM.CIÊ¯. UNIWERSALNY' then 'SAMOCHÓD CIÊ¯AROWY'
else p.rodzaj_pojazdu end as rodzaj
from pojazdy p
where p.rodzaj_paliwa='ENERGIA ELEKTRYCZNA'
and rodzaj_pojazdu not in ('PRZYCZEPA CIÊ¯AROWA', 'PRZYCZEPA CIÊ¯AROWA ROLNICZA', 'PRZYCZEPA LEKKA', 'PRZYCZEPA SPECJALNA','POJAZD WOLNOBIE¯NY-KOLEJKA TURYSTYCZNA')
group by extract(year from data_ostatniej_rejestracji_w_kraju),
case p.rodzaj_pojazdu when 'CI¥GNIK SAMOCHODOWY' THEN 'SAMOCHÓD CIÊ¯AROWY' when 'SAMOCHÓD SPECJALNY' then 'SAMOCHÓD CIÊ¯AROWY' when 'SAMOCHODOWY INNY' THEN 'SAMOCHÓD CIÊ¯AROWY' when 'SAM.CIÊ¯. UNIWERSALNY' then 'SAMOCHÓD CIÊ¯AROWY'
else p.rodzaj_pojazdu end

CREATE TABLE favorite_pools (
    id BIGSERIAL PRIMARY KEY,
    pool_id BIGINT NOT NULL,
    user_id VARCHAR NOT NULL,
    UNIQUE (pool_id, user_id)
);

CREATE TABLE user_logs (
    id SERIAL PRIMARY KEY,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    user_oid VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    name VARCHAR(512) NOT NULL
);


CREATE TABLE IF NOT EXISTS projects (
    id UUID PRIMARY KEY,
    name VARCHAR(512) NOT NULL,
    description TEXT,
    location GEOGRAPHY(Point, 4326) NOT NULL,
    operator_id INTEGER NOT NULL, 
    user_id VARCHAR(255) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_projects_location ON projects USING GIST(location);
CREATE INDEX IF NOT EXISTS idx_projects_user_id ON projects(user_id);

CREATE TABLE pojazdy_powiaty_summary (
  nazwa_powiatu text DEFAULT NULL,
  kod_woj_powiat integer DEFAULT NULL,
  liczba_bev integer DEFAULT NULL,
  liczba_wszystkich integer DEFAULT NULL
);

CREATE TABLE IF NOT EXISTS projects_points (
  project_id UUID NOT NULL,
  power integer NOT NULL,
  price integer NOT NULL
);
ALTER TABLE projects_points ADD FOREIGN KEY (project_id) REFERENCES projects (id) ON DELETE CASCADE;



CREATE TABLE user_settings (
    user_oid VARCHAR(255) NOT NULL PRIMARY KEY,
    operator_id integer DEFAULT NULL
);

ALTER TABLE pools ADD usage_percentage NUMERIC DEFAULT NULL

CREATE TABLE IF NOT EXISTS pools_summary (
    pool_code text DEFAULT NULL,
    pool_point_count INT NOT NULL DEFAULT 0,
    charging_total BIGINT NOT NULL DEFAULT 0,
    availability_total BIGINT NOT NULL DEFAULT 0,
    usage_percentage NUMERIC(9, 6) NOT NULL DEFAULT 0.000000
);
